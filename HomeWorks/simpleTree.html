Что такое бинарное дерево?

Бинарное дерево — это структура данных, в которой каждый узел имеет не более двух потомков: левого и правого. Оно используется в алгоритмах поиска, сортировки и других задачах.

Принцип работы:

  • Каждый узел содержит данные и две ссылки:
    • left — указывает на левого потомка.
    • right — указывает на правого потомка.
  • Добавление элемента определяется его значением: меньше родителя — влево, больше — вправо.
  • Обход дерева возможен в различных порядках: прямой, симметричный и обратный.

Код задания


// Узел бинарного дерева
class TreeNode {
    var value: T
    var left: TreeNode?
    var right: TreeNode?

    init(value: T) {
        self.value = value
    }
}

// Бинарное дерево
class BinaryTree {
    var root: TreeNode?
    
    // Добавление элемента
    func insert(_ value: T) {
        root = insertRecursively(root, value)
    }
    
    private func insertRecursively(_ node: TreeNode?, _ value: T) -> TreeNode {
        guard let node = node else { return TreeNode(value: value) }
        
        if value < node.value {
            node.left = insertRecursively(node.left, value)
        } else {
            node.right = insertRecursively(node.right, value)
        }
        return node
    }
    
    // Обход дерева в порядке "In-Order"
    func inOrderTraversal(_ node: TreeNode?) {
        guard let node = node else { return }
        inOrderTraversal(node.left)
        print(node.value)
        inOrderTraversal(node.right)
    }
}

// Пример использования
let tree = BinaryTree()
tree.insert(5)
tree.insert(3)
tree.insert(7)
tree.insert(2)
tree.insert(4)
tree.insert(6)
tree.insert(8)

tree.inOrderTraversal(tree.root) // Выведет: 2 3 4 5 6 7 8
